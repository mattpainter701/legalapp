```python
# backend/app/reporting/queries.py
from sqlalchemy import func, select, case
from sqlalchemy.orm import Session
from app.models import Matter, Client, TimeEntry, Invoice, InvoiceLineItem, Payment

def realization_report(session: Session):
    """
    Returns list of dicts: matter_id, matter_name, client_name, status,
    hours_billed, amount_billed, amount_paid, realization_pct
    """
    query = select(
        Matter.id.label('matter_id'),
        Matter.name.label('matter_name'),
        Client.name.label('client_name'),
        Matter.status.label('status'),
        func.coalesce(func.sum(TimeEntry.hours), 0).label('hours_billed'),
        func.coalesce(func.sum(InvoiceLineItem.amount), 0).label('amount_billed'),
        func.coalesce(func.sum(Payment.amount), 0).label('amount_paid'),
    ).select_from(Matter).join(
        Client, Matter.client_id == Client.id
    ).outerjoin(
        TimeEntry, TimeEntry.matter_id == Matter.id
    ).outerjoin(
        Invoice, Invoice.matter_id == Matter.id
    ).outerjoin(
        InvoiceLineItem, InvoiceLineItem.invoice_id == Invoice.id
    ).outerjoin(
        Payment, Payment.invoice_id == Invoice.id
    ).group_by(
        Matter.id, Matter.name, Client.name, Matter.status
    )
    
    rows = session.execute(query).fetchall()
    
    result = []
    for row in rows:
        amount_billed = row.amount_billed or 0
        amount_paid = row.amount_paid or 0
        realization_pct = (amount_paid / amount_billed * 100) if amount_billed > 0 else 0
        
        result.append({
            'matter_id': row.matter_id,
            'matter_name': row.matter_name,
            'client_name': row.client_name,
            'status': row.status,
            'hours_billed': float(row.hours_billed),
            'amount_billed': float(amount_billed),
            'amount_paid': float(amount_paid),
            'realization_pct': round(realization_pct, 2)
        })
    
    return result
```

```python
def wip_report(session: Session):
    """Work-in-progress: uninvoiced TimeEntry hours and estimated value"""
    query = select(
        Matter.id.label('matter_id'),
        Matter.name.label('matter_name'),
        Client.name.label('client_name'),
        func.coalesce(func.sum(TimeEntry.hours), 0).label('wip_hours'),
        func.coalesce(func.sum(TimeEntry.hours * TimeEntry.rate), 0).label('wip_value'),
    ).select_from(Matter).join(
        Client, Matter.client_id == Client.id
    ).outerjoin(
        TimeEntry, (TimeEntry.matter_id == Matter.id) & (TimeEntry.invoiced == False)
    ).group_by(
        Matter.id, Matter.name, Client.name
    )
    
    rows = session.execute(query).fetchall()
    return [
        {
            'matter_id': row.matter_id,
            'matter_name': row.matter_name,
            'client_name': row.client_name,
            'wip_hours': float(row.wip_hours),
            'wip_value': float(row.wip_value)
        }
        for row in rows
    ]

def ar_aging_report(session: Session):
    """Accounts receivable aging: unpaid invoices by days overdue"""
    today = datetime.now().date()
    
    query = select(
        Matter.id.label('matter_id'),
        Matter.name.label('matter_name'),
        Client.name.label('client_name'),
        func.coalesce(func.sum(
            case((func.julianday(today) - func.julianday(Invoice.date) <= 30, Invoice.amount), else_=0)
        ), 0).label('days_0_30'),
        func.coalesce(func.sum(
            case(((func.julianday(today) - func.julianday(Invoice.date) > 30) & 
                  (func.julianday(today) - func.julianday(Invoice.date) <= 60), Invoice.amount), else_=0)
        ), 0).label('days_31_60'),
        func.coalesce(func.sum(
            case(((func.julianday(today) - func.julianday(Invoice.date) > 60) & 
                  (func.julianday(today) - func.julianday(Invoice.date) <= 90), Invoice.amount), else_=0)
        ), 0).label('days_61_90'),
        func.coalesce(func.sum(
            case((func.julianday(today) - func.julianday(Invoice.date) > 90, Invoice.amount), else_=0)
        ), 0).label('days_90_plus'),
    ).select_from(Matter).join(
        Client, Matter.client_id == Client.id
    ).outerjoin(
        Invoice, (Invoice.matter_id == Matter.id) & (Invoice.status == 'open')
    ).group_by(
        Matter.id, Matter.name, Client.name
    )
    
    rows = session.execute(query).fetchall()
    return [
        {
            'matter_id': row.matter_id,
            'matter_name': row.matter_name,
            'client_name': row.client_name,
            'days_0_30': float(row.days_0_30),
            'days_31_60': float(row.days_31_60),
            'days_61_90': float(row.days_61_90),
            'days_90_plus': float(row.days_90_plus),
        }
        for row in rows
    ]
```

```python
# backend/app/routes/reports.py
import csv
import io
from fastapi import APIRouter, Depends, Query
from sqlalchemy.orm import Session
from app.database import get_db
from app.reporting.queries import realization_report, wip_report, ar_aging_report
from fastapi.responses import StreamingResponse

router = APIRouter(prefix="/reports", tags=["reports"])

@router.get("/realization")
def get_realization_report(
    session: Session = Depends(get_db),
    format: str = Query("json", regex="^(json|csv)$")
):
    """Get realization report: revenue recognized by matter"""
    data = realization_report(session)
    
    if format == "csv":
        return _to_csv_response(data, "realization_report.csv")
    return data

@router.get("/wip")
def get_wip_report(
    session: Session = Depends(get_db),
    format: str = Query("json", regex="^(json|csv)$")
):
    """Get WIP report: uninvoiced hours by matter"""
    data = wip_report(session)
    
    if format == "csv":
        return _to_csv_response(data, "wip_report.csv")
    return data

@router.get("/ar-aging")
def get_ar_aging_report(
    session: Session = Depends(get_db),
    format: str = Query("json", regex="^(json|csv)$")
):
    """Get AR aging report: outstanding invoices by days overdue"""
    data = ar_aging_report(session)
    
    if format == "csv":
        return _to_csv_response(data, "ar_aging_report.csv")
    return data

def _to_csv_response(data, filename):
    """Helper: convert list of dicts to CSV StreamingResponse"""
    if not data:
        return StreamingResponse(iter([""]), media_type="text/csv")
    
    output = io.StringIO()
    writer = csv.DictWriter(output, fieldnames=data[0].keys())
    writer.writeheader()
    writer.writerows(data)
    
    output.seek(0)
    return StreamingResponse(
        iter([output.getvalue()]),
        media_type="text/csv",
        headers={"Content-Disposition": f"attachment; filename={filename}"}
    )
```

```javascript
// frontend/src/components/ReportTable.jsx
import React, { useState } from 'react';
import { ArrowUp, ArrowDown, Download } from 'lucide-react';

const ReportTable = ({ data, columns, title, onDownloadCSV }) => {
  const [sortKey, setSortKey] = useState(null);
  const [sortDirection, setSortDirection] = useState('asc');

  const handleSort = (key) => {
    if (sortKey === key) {
      setSortDirection(sortDirection === 'asc' ? 'desc' : 'asc');
    } else {
      setSortKey(key);
      setSortDirection('asc');
    }
  };

  const sortedData = React.useMemo(() => {
    if (!sortKey || !data) return data || [];
    
    const sorted = [...data].sort((a, b) => {
      const aVal = a[sortKey];
      const bVal = b[sortKey];
      
      if (typeof aVal === 'number') {
        return sortDirection === 'asc' ? aVal - bVal : bVal - aVal;
      }
      return sortDirection === 'asc'
        ? String(aVal).localeCompare(String(bVal))
        : String(bVal).localeCompare(String(aVal));
    });
    return sorted;
  }, [data, sortKey, sortDirection]);

  if (!data || data.length === 0) {
    return <div className="p-4 text-gray-500">No data available</div>;
  }

  return (
    <div className="space-y-4">
      <div className="flex justify-between items-center">
        <h2 className="text-lg font-semibold">{title}</h2>
        <button
          onClick={onDownloadCSV}
          className="flex items-center gap-2 px-3 py-2 bg-blue-600 text-white rounded hover:bg-blue-700"
        >
          <Download size={16} />
          Download CSV
        </button>
      </div>

      <table className="w-full border-collapse border border-gray-300">
        <thead className="bg-gray-100">
          <tr>
            {columns.map((col) => (
              <th
                key={col.key}
                onClick={() => handleSort(col.key)}
                className="border border-gray-300 p-3 text-left cursor-pointer hover:bg-gray-200"
              >
                <div className="flex items-center gap-2">
                  {col.label}
                  {sortKey === col.key && (
                    sortDirection === 'asc' ? <ArrowUp size={14} /> : <ArrowDown size={14} />
                  )}
                </div>
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {sortedData.map((row, i) => (
            <tr key={i} className={i % 2 === 0 ? 'bg-white' : 'bg-gray-50'}>
              {columns.map((col) => (
                <td key={col.key} className="border border-gray-300 p-3">
                  {typeof row[col.key] === 'number'
                    ? col.format
                      ? col.format(row[col.key])
                      : row[col.key].toFixed(2)
                    : row[col.key]}
                </td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
};

export default ReportTable;
```

```javascript
    { key: 'matter_name', label: 'Matter' },
    { key: 'client_name', label: 'Client' },
    { key: 'wip_hours', label: 'WIP Hours', format: (v) => v.toFixed(1) },
    { key: 'wip_value', label: 'WIP Value', format: (v) => `$${v.toFixed(2)}` },
  ];

  const arAgingColumns = [
    { key: 'matter_name', label: 'Matter' },
    { key: 'client_name', label: 'Client' },
    { key: 'days_0_30', label: '0-30 Days', format: (v) => `$${v.toFixed(2)}` },
    { key: 'days_31_60', label: '31-60 Days', format: (v) => `$${v.toFixed(2)}` },
    { key: 'days_61_90', label: '61-90 Days', format: (v) => `$${v.toFixed(2)}` },
    { key: 'days_90_plus', label: '90+ Days', format: (v) => `$${v.toFixed(2)}` },
  ];

  return (
    <div className="p-6">
      <h1 className="text-3xl font-bold mb-6">Accounting Reports</h1>

      <div className="flex gap-2 border-b mb-6">
        {[
          { id: 'realization', label: 'Realization' },
          { id: 'wip', label: 'WIP' },
          { id: 'ar-aging', label: 'A/R Aging' },
        ].map((tab) => (
          <button
            key={tab.id}
            onClick={() => setActiveTab(tab.id)}
            className={`px-4 py-2 font-medium ${
              activeTab === tab.id
                ? 'border-b-2 border-blue-600 text-blue-600'
                : 'text-gray-600 hover:text-gray-900'
            }`}
          >
            {tab.label}
          </button>
        ))}
      </div>

      <div className="bg-white rounded shadow">
        {activeTab === 'realization' && (
          <div className="p-6">
            <ReportTable
              data={realization}
              columns={realizationColumns}
              title="Realization Report"
              onDownloadCSV={() =>
                handleDownloadCSV(downloadRealizationCSV, 'realization_report.csv')
              }
            />
          </div>
        )}

        {activeTab === 'wip' && (
          <div className="p-6">
            <ReportTable
              data={wip}
              columns={wipColumns}
              title="Work in Progress"
              onDownloadCSV={() => handleDownloadCSV(downloadWIPCSV, 'wip_report.csv')}
            />
          </div>
        )}

        {activeTab === 'ar-aging' && (
          <div className="p-6">
            <ReportTable
              data={arAging}
              columns={arAgingColumns}
              title="Accounts Receivable Aging"
              onDownloadCSV={() =>
                handleDownloadCSV(downloadARAgingCSV, 'ar_aging_report.csv')
              }
            />
          </div>
        )}
      </div>
    </div>
  );
};

export default ReportsPage;
```

# Accounting Reports API

## Endpoints

### GET /reports/realization

Returns revenue realization data by matter.

**Query Parameters:**
- `format` (string, optional): `json` (default) or `csv`

```json
    "realization_pct": 75.0
  }
]
```

### GET /reports/wip

Returns work-in-progress (uninvoiced) hours and estimated value by matter.

**Query Parameters:**
- `format` (string, optional): `json` (default) or `csv`

**Response (JSON):**
```json
[
  {
    "matter_id": 1,
    "matter_name": "Litigation Case",
    "client_name": "Acme Corp",
    "wip_hours": 12.5,
    "wip_value": 2500.00
  }
]
```

Returns outstanding invoices segmented by days overdue.

- `format` (string, optional): `json` (default) or `csv`

[
